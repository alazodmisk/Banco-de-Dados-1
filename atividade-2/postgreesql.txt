CREATE TABLE Cliente 
( 
      idCliente INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,  
      Nome VARCHAR(100) NOT NULL,  
      Idade INT NOT NULL  
); 


CREATE TABLE Locacao 
(
    IdLocacao INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Valor_diaria NUMERIC(10,2) NOT NULL,
    Km_percorridos NUMERIC(10,2) NOT NULL,
    dias_locados INT NOT NULL,
    Valor_total NUMERIC(10,2) NOT NULL,
    idCliente INT,

    CONSTRAINT fk_Locacao_cliente
        FOREIGN KEY (idCliente)
        REFERENCES Cliente(idCliente)
        ON DELETE RESTRICT
);


CREATE TABLE Seguro 
( 
    IdSeguro INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, 
    Valor_seguro FLOAT NOT NULL,  
    IdLocacao INT,

    CONSTRAINT fk_Locacao_seguro
        FOREIGN KEY (IdLocacao)
        REFERENCES Locacao(IdLocacao)
        ON DELETE RESTRICT
);

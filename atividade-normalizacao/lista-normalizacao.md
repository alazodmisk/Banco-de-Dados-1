##### EXERCICIO 6)
$A-$ Para cada nova disciplina que um aluno é matriculado, uma nova coluna deverá ser criada

$B-$ Isso ocorre pois viola as regras de normalização (NF1). Desse modo, o banco de dados fica mais complexo e ineficiente em consultas

$C-$ 2 novas tabelas precisariam ser criadas, apenas um aluno usaria porém afetaria todos os alunos

$D-$ 1. Modelo Lógico Relacional Proposto
    
    ALUNO
        id_aluno (PK)
        nome_aluno

    DISCIPLINA
        id_disciplina (PK)
        nome_disciplina

    ALUNO_DISCIPLINA (ou MATRICULA)
        id_aluno (FK - Referencia ALUNO.id_aluno)
        id_disciplina (FK - Referencia DISCIPLINA.id_disciplina)
        (Composta por uma Chave Primária Dupla: id_aluno + id_disciplina)

$E-$ ALUNO (1) ----- (N) ALUNO_DISCIPLINA (N) ----- (1) DISCIPLINA  

$F-$

    CREATE TABLE aluno (
        id_aluno INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        nome_aluno VARCHAR(100) NOT NULL
    );

    CREATE TABLE disciplina (
        id_disciplina INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
        nome_disciplina VARCHAR(100) NOT NULL
    );

    CREATE TABLE aluno_disciplina (
        id_aluno INT NOT NULL,
        id_disciplina INT NOT NULL,
        
        PRIMARY KEY (id_aluno, id_disciplina),
        
        CONSTRAINT fk_aluno
            FOREIGN KEY (id_aluno)
            REFERENCES aluno(id_aluno)
            ON DELETE CASCADE,
            
        CONSTRAINT fk_disciplina
            FOREIGN KEY (id_disciplina)
            REFERENCES disciplina(id_disciplina)
            ON DELETE CASCADE
    );
